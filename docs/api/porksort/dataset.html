porksort.dataset

Reading the Roboflow YOLOv8 export and building an image index.

The export has one .txt label file per image. Each line is one bounding box in normalised YOLO format::

<class_id> <x_centre> <y_centre> <width> <height>

with all four coordinates in [0, 1] relative to the image size. An empty file means the annotator found no defect, which this project treats as a PASS frame.

Roboflow names augmented copies <source>.rf.<hash>.jpg. All copies of one source frame must stay in the same split, otherwise the model is tested on a brightness-shifted copy of an image it trained on. source_id recovers the source name so the splitting code can group copies.

 75def parse_label_file(path: str | os.PathLike) -> list[Box]:
 76    """Parse one YOLO label file.
 77
 78    Args:
 79        path: Path to a ``.txt`` label file.
 80
 81    Returns:
 82        The boxes in the file, possibly an empty list (a PASS frame).
 83
 84    Raises:
 85        FileNotFoundError: If the file does not exist.
 86        ValueError: If a line does not have 5 fields, the class id is not a
 87            known class, or a coordinate is outside [0, 1]. The message names
 88            the file and line so a bad annotation can be found and fixed.
 89    """
 90    boxes: list[Box] = []
 91    text = Path(path).read_text()
 92    for lineno, line in enumerate(text.splitlines(), start=1):
 93        parts = line.split()
 94        # Blank lines are allowed; some export tools end the file with one.
 95        if not parts:
 96            continue
 97        if len(parts) != 5:
 98            raise ValueError(f"{path}:{lineno}: expected 5 fields, got {len(parts)}")
 99        try:
100            cid = int(parts[0])
101            xc, yc, w, h = (float(p) for p in parts[1:])
102        except ValueError as exc:
103            raise ValueError(f"{path}:{lineno}: non-numeric field") from exc
104        # Validate here, so a bad annotation fails loudly with its file and line
105        # instead of silently becoming a wrong label later.
106        if not 0 <= cid < len(CLASS_NAMES):
107            raise ValueError(f"{path}:{lineno}: unknown class id {cid}")
108        for name, v in (("xc", xc), ("yc", yc), ("w", w), ("h", h)):
109            if not 0.0 <= v <= 1.0:
110                raise ValueError(f"{path}:{lineno}: {name}={v} outside [0, 1]")
111        boxes.append(Box(CLASS_NAMES[cid], xc, yc, w, h))
112    return boxes

115def source_id(filename: str) -> str:
116    """Return the source-frame name shared by all augmented copies.
117
118    Args:
119        filename: Image file name or path.
120
121    Returns:
122        The part before ``.rf.`` for a Roboflow copy, otherwise the file stem.
123
124    Examples:
125        >>> source_id("frame_with_red_10049_jpg.rf.cd786c4f.jpg")
126        'frame_with_red_10049_jpg'
127        >>> source_id("my_photo.jpg")
128        'my_photo'
129    """
130    stem = Path(filename).name
131    if ".rf." in stem:
132        return stem.split(".rf.")[0]
133    return Path(stem).stem

136def frame_number(source: str) -> int | None:
137    """Extract the video frame number from a source name.
138
139    Args:
140        source: Source-frame name from `source_id`.
141
142    Returns:
143        The frame number, or None if the name does not follow the export's
144        ``frame_with_red_<n>_jpg`` pattern.
145
146    Examples:
147        >>> frame_number("frame_with_red_10049_jpg")
148        10049
149        >>> frame_number("other") is None
150        True
151    """
152    m = _FRAME_RE.search(source)
153    return int(m.group(1)) if m else None

168def build_index(root: str | os.PathLike | None = None) -> pd.DataFrame:
169    """Index every image in the export, one row per image file.
170
171    Args:
172        root: Folder that contains ``train/``, ``valid/`` and ``test/``.
173            Defaults to the ``PORKSORT_DATA`` environment variable, then
174            `porksort.config.DEFAULT_DATA_DIR`.
175
176    Returns:
177        DataFrame with columns ``rf_split``, ``image``, ``label``, ``source``,
178        ``frame``, ``copy`` (0, 1, 2 ... within a source), ``n_boxes``,
179        ``fail`` (bool), ``classes`` ("|"-joined class names present) and one
180        count column ``n_<class>`` per class. Rows are sorted by split then
181        filename, so the index is deterministic.
182
183    Raises:
184        FileNotFoundError: If ``root`` has none of the expected split folders,
185            or an image has no matching label file.
186    """
187    root = Path(root or os.environ.get("PORKSORT_DATA", DEFAULT_DATA_DIR))
188    rows = []
189    found_any = False
190    for split in SPLITS:
191        img_dir, lbl_dir = root / split / "images", root / split / "labels"
192        if not img_dir.is_dir():
193            continue
194        found_any = True
195        for img in sorted(img_dir.iterdir()):
196            # Skip hidden files such as macOS ._frame.jpg metadata.
197            if img.name.startswith("."):
198                continue
199            lbl = lbl_dir / (img.stem + ".txt")
200            if not lbl.exists():
201                raise FileNotFoundError(f"no label file for {img}")
202            boxes = parse_label_file(lbl)
203            src = source_id(img.name)
204            row = {
205                "rf_split": split,
206                "image": str(img),
207                "label": str(lbl),
208                "source": src,
209                "frame": frame_number(src),
210                "n_boxes": len(boxes),
211                "fail": is_fail(boxes),
212                "classes": "|".join(sorted({b.cls for b in boxes})),
213            }
214            for c in CLASS_NAMES:
215                row[f"n_{c}"] = sum(b.cls == c for b in boxes)
216            rows.append(row)
217    if not found_any:
218        raise FileNotFoundError(
219            f"no train/valid/test folders under {root}. Unzip the Roboflow export there or set PORKSORT_DATA."
220        )
221    df = pd.DataFrame(rows)
222    # Number the augmented copies of each source frame 0, 1, 2 ...; evaluation keeps copy 0 only.
223    df["copy"] = df.groupby("source").cumcount()
224    return df

227def unique_sources(index: pd.DataFrame) -> pd.DataFrame:
228    """Keep one row per source frame, for splitting and audits.
229
230    Args:
231        index: Output of `build_index`.
232
233    Returns:
234        The first copy (by file name) of each source frame, re-indexed from 0.
235    """
236    return index.sort_values(["source", "image"]).drop_duplicates("source").reset_index(drop=True)
